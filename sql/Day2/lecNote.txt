************************ Groub By && Having *********************
Group by : group rows with  same value    ==> AAggregiation functions (sum , avg , min , max , count)
Having : filteration for group by 
Where : condition before group by 


--select num of employees in each department 

Select dep_num, count(*) as employee_count from employees GROUP by (dep_num);

-- select total salary for employees in each department4
Select dep_num, sum(salary) as total_salry from employees GROUP by (dep_num);

Having :  filteration of groub by ===> condition depend on value of having

-- select  depertment which its number of employees more than 2 
** numder of employees in each department + department count (employees) >2

Select dep_num, count(*) as employee_count from employees GROUP by (dep_num) having count(*)>2

-- select departments that total salry of employees > 12000
Select dep_num, sum(salary) as salary_total from employees GROUP by (dep_num) having sum(salary)>12000;

--------------- innner join  (particepated data)   -----------

** select department name for each employee

select employees.name as employee_name , employees.email , departments.name as department_name from employees 
INNER join departments
 WHERE employees.dep_num = departments.num;

-----------   alias table -----------
select e.name as employee_name , e.email , d.name as department_name 
from employees  e
INNER join departments d
 WHERE e.dep_num = d.num;


** select manager name for each department 
SELECT d.num , d.name as department_name , e.name as manger_name from departments d INNER join employees e where d.manger_id=e.SSN;


------------------- left join  : get data from left table
employees   left join     departments

select all departments even it hasn't any employees

Select e.name as employee_name , e.email , d.name as department_name from employees e
RIGHT JOIN  departments d 
ON e.dep_num = d.num;
